
Agent不会产出StopMesssage，StopMessage都是Manager创建并且都会调用_signal_termination把它装进GroupChatTermination，也就是能进Message队列的都是GroupChatTermination。而StopMessageTermination说是如果收到StopMessage就终止，它只检测是否是StopMessage，但根据上述推理，传入StopMessageTermination的message根本不会是StopMessage，他有什么用


class GroupManager
    handle_agent_response{
        if(_apply_termination_condition) retrun
        else _transition_to_next_speakers()
    }

    _apply_termination_condition(message){
        if self._termination_condition is not None: return;
            await self._termination_condition.reset()
            self._current_turn = 0
            await self._signal_termination(stop_message)
        if self._current_turn >= self._max_turns:
            stop_message = StopMessage(
                content=f"Maximum number of turns {self._max_turns} reached.",
                source=self._name,
            )
    }

    _transition_to_next_speakers(){
        await self.publish_message(
            GroupChatRequestPublish(),
            topic_id=DefaultTopicId(type=speaker_topic_type),
            cancellation_token=cancellation_token,
        )
    }

    async def _signal_termination(self, message: StopMessage) -> None:
        await self.publish_message(
            GroupChatTermination(message=message),
            topic_id=DefaultTopicId(type=self._output_topic_type)
        )
